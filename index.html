─── Constants ──────────────────────────────────────────────────────────────── TRIG_PIN = 23 # GPIO pin for HC-SR04 trigger ECHO_PIN = 24 # GPIO pin for HC-SR04 echo BUZZER_PIN = 17 # GPIO buzzer pin SPEED_OF_SOUND = 34300 # ─── Ultrasonic Distance Measurement ────────────────────────────────────────── def measure_distance_ultrasonic(): """Measure distance using HC-SR04 ultrasonic sensor.""" if not GPIO_AVAILABLE: return None GPIO.output(TRIG_PIN, True) time.sleep(0.00001) GPIO.output(TRIG_PIN, False) pulse_start = time.time() pulse_end = time.time() timeout = time.time() + 0.04 # ─── Optical Flow Speed Estimator ───────────────────────────────────────────── class SpeedEstimator: """ Estimates object/vehicle speed using Lucas-Kanade Optical Flow. Works with a standard webcam. Tracks feature points between consecutive frames and converts pixel displacement to real-world speed (km/h). """ def __init__(self, fps=30, ppm=PIXELS_PER_METER): self.fps = fps self.ppm = ppm # pixels per meter self.prev_gray = None self.prev_pts = None self.speed_kmh = 0.0 self.speed_history = [] self.frame_count = 0 # Lucas-Kanade optical flow parameters self.lk_params = dict( winSize = (21, 21), maxLevel = 3, criteria = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 30, 0.01) ) # Shi-Tomasi corner detection parameters self.feature_params = dict( maxCorners = 100, qualityLevel = 0.3, minDistance = 7, blockSize = 7 ) def estimate(self, frame): """Process a frame and return estimated speed in km/h.""" small = cv2.resize(frame, (0, 0), fx=FRAME_SCALE, fy=FRAME_SCALE) gray = cv2.cvtColor(small, cv2.COLOR_BGR2GRAY) self.frame_count += 1 if self.prev_gray is None: self.prev_gray = gray self.prev_pts = cv2.goodFeaturesToTrack(gray, mask=None, **self.feature_params) return 0.0, frame if self.prev_pts is None or len(self.prev_pts) < 5: self.prev_pts = cv2.goodFeaturesToTrack(gray, mask=None, **self.feature_params) self.prev_gray = gray return self.speed_kmh, frame # Calculate optical flow next_pts, status, _ = cv2.calcOpticalFlowPyrLK( self.prev_gray, gray, self.prev_pts, None, **self.lk_params ) if next_pts is None: self.prev_gray = gray return self.speed_kmh, frame good_new = next_pts[status == 1] good_prev = self.prev_pts[status == 1] if len(good_new) < 3: self.prev_pts = cv2.goodFeaturesToTrack(gray, mask=None, **self.feature_params) self.prev_gray = gray return self.speed_kmh, frame # Compute displacements displacements = np.linalg.norm(good_new - good_prev, axis=1) median_disp = np.median(displacements) # pixels per frame # Convert: pixels/frame → m/s → km/h meters_per_frame = median_disp / (self.ppm * FRAME_SCALE) meters_per_sec = meters_per_frame * self.fps kmh = meters_per_sec * 3.6 # Smooth speed over last 10 readings self.speed_history.append(kmh) if len(self.speed_history) > 10: self.speed_history.pop(0) self.speed_kmh = round(np.mean(self.speed_history), 1) # Draw flow vectors on frame mask = np.zeros_like(frame) for new, prev in zip(good_new / FRAME_SCALE, good_prev / FRAME_SCALE): a, b = new.ravel().astype(int) c, d = prev.ravel().astype(int) cv2.arrowedLine(frame, (c, d), (a, b), (0, 255, 100), 2, tipLength=0.4) cv2.circle(frame, (a, b), 3, (0, 200, 255), -1) # Refresh feature points every 15 frames if self.frame_count % 15 == 0: self.prev_pts = cv2.goodFeaturesToTrack(gray, mask=None, **self.feature_params) else: self.prev_pts = good_new.reshape(-1, 1, 2) self.prev_gray = gray return self.speed_kmh, frame # ─── Camera-Based Distance Estimator ────────────────────────────────────────── class DistanceEstimator: """ Estimates distance to rear obstacle using contour detection. Falls back to ultrasonic sensor if GPIO is available. Principle: uses apparent width of detected objects vs known real width. Distance = (Known_Width * Focal_Length) / Perceived_Width_in_pixels """ def __init__(self, focal_length=FOCAL_LENGTH, known_width_cm=30): self.focal_length = focal_length self.known_width_cm = known_width_cm # real width of reference object self.distance_cm = 999.0 self.dist_history = [] def estimate(self, frame): """Return estimated distance in cm and annotated frame.""" # Prefer hardware ultrasonic sensor hw_dist = measure_distance_ultrasonic() if hw_dist is not None: self.distance_cm = hw_dist return hw_dist, frame # Software fallback: detect largest contour in ROI h, w = frame.shape[:2] roi = frame[h//2:, :] # bottom half = rear camera view gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY) blur = cv2.GaussianBlur(gray, (7, 7), 0) edges = cv2.Canny(blur, 50, 150) contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE) if contours: largest = max(contours, key=cv2.contourArea) if cv2.contourArea(largest) > 500: x, y, cw, ch = cv2.boundingRect(largest) if cw > 20: dist = (self.known_width_cm * self.focal_length) / cw dist = round(dist, 1) # Smooth self.dist_history.append(dist) if len(self.dist_history) > 8: self.dist_history.pop(0) self.distance_cm = round(np.mean(self.dist_history), 1) # Draw bounding box on frame ry = y + h // 2 cv2.rectangle(frame, (x, ry), (x + cw, ry + ch), (0, 120, 255), 2) cv2.putText(frame, f"{self.distance_cm} cm", (x, ry - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 120, 255), 2) return self.distance_cm, frame # ══════════════════════════════════════════════════════════════════════════ # SYSTEM CONTROL # ══════════════════════════════════════════════════════════════════════════ def toggle_system(self): if not self.running: self.running = True self.start_btn.config(text="⏹ STOP SYSTEM", bg="#330000", fg="#FF4444") self.status_lbl.config(text="● ONLINE", fg="#00FF88") self._log("System STARTED.") if not self.sim_mode.get(): self._start_camera() else: self._log("Running in SIMULATION mode.") else: self.running = False self.start_btn.config(text="▶ START SYSTEM", bg="#003322", fg="#00FF88") self.status_lbl.config(text="● OFFLINE", fg="#FF4444") self._stop_camera() self._log("System STOPPED.") def _start_camera(self): self.cap = cv2.VideoCapture(0) if not self.cap.isOpened(): self._log("WARNING: Camera not found. Switching to simulation.") self.sim_mode.set(True) self.cap = None return self.camera_active = True self._log("Camera opened successfully.") t = threading.Thread(target=self._camera_thread, daemon=True) t.start() def _stop_camera(self): self.camera_active = False if self.cap: self.cap.release() self.cap = None def _camera_thread(self): """Background thread: captures frames, estimates speed & distance.""" fps_time = time.time() frame_count = 0 while self.camera_active and self.running: ret, frame = self.cap.read() if not ret: continue frame_count += 1 elapsed = time.time() - fps_time if elapsed > 0: fps = frame_count / elapsed self.speed_est.fps = max(1, fps) # Estimate speed & distance speed, annotated = self.speed_est.estimate(frame.copy()) dist, annotated = self.dist_est.estimate(annotated) # Put in queue (non-blocking) if not self.frame_queue.full(): frame_queue_item = (speed, dist, annotated) self.frame_queue.put(frame_queue_item) def _simulate_step(self): """Generate realistic-looking simulation data.""" # Simulate distance: obstacle approaching/receding self._sim_dist += random.uniform(-8, 6) self._sim_dist = max(20, min(400, self._sim_dist)) return round(self._sim_speed, 1), round(self._sim_dist, 1) # ALERT LOGIC # ══════════════════════════════════════════════════════════════════════════ def _compute_alert(self, speed, dist): if dist <= DANGER_DIST: self.alert_count += 1 self._log(f"⚠ DANGER! ══════════════════════════════════════════════════════════════════════════ # GAUGE DRAWING # ══════════════════════════════════════════════════════════════════════════ def _draw_gauge(self, canvas, value, max_val, unit, arc_color, danger_thresh=None, warn_thresh=None): """Generic arc-style gauge.""" canvas.delete("all") w = canvas.winfo_width() h = canvas.winfo_height() if w < 10 or h < 10: return cx = w // 2 cy = int(h * 0.60) r = min(cx, cy) - 12 start = 210 # degrees span = 300 # degrees arc # Background arc canvas.create_arc(cx-r, cy-r, cx+r, cy+r, start=start, extent=-span, outline="#1A2433", style="arc", width=12) # Colored arc (value) ratio = min(value / max_val, 1.0) angle = span * ratio # Determine color if danger_thresh and value >= danger_thresh: color = "#FF2200" elif warn_thresh and value >= warn_thresh: color = "#FFAA00" else: color = arc_color if angle > 0: canvas.create_arc(cx-r, cy-r, cx+r, cy+r, start=start, extent=-angle, outline=color, style="arc", width=12, dash=() ) #
